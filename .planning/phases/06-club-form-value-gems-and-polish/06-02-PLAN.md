```typescript
import { readFile } from 'fs/promises'
import { join } from 'path'

export async function GET() {
  try {
    const cachePath = join(process.cwd(), 'pipeline', 'cache', 'defcon_stats.json')
    const data = await readFile(cachePath, 'utf-8')
    return new Response(data, {
      status: 200,
      headers: {
        'Content-Type': 'application/json',
        'Cache-Control': 'public, s-maxage=3600, stale-while-revalidate=86400',
      },
    })
  } catch {
    return Response.json({ error: 'DefCon data not available' }, { status: 404 })
  }
}
```

6. Create `src/components/LastUpdated.tsx`:

```typescript
'use client'

import { useLastUpdated } from '@/lib/hooks/useLastUpdated'

export function LastUpdated() {
  const { data } = useLastUpdated()
  if (!data) return null

  const d = new Date(data.last_updated)
  const label = d.toLocaleString('en-GB', { dateStyle: 'short', timeStyle: 'short' })

  return (
    <p className={`text-xs mt-1 ${data.stale ? 'text-amber-600' : 'text-zinc-400'}`}>
      Data as of {label}{data.stale ? ' (stale)' : ''}
    </p>
  )
}
```
