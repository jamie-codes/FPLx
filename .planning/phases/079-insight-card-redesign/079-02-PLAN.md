  // Existing 6 fields (Phase 33; preserved for backwards compatibility per D-03)

  sample_n: number                                        // numerator (how many times pattern held true)
  sample_total: number                                    // denominator (>= 10 enforced by pipeline)
  // Phase 79 D-01 — structured display fields
  title: string                                           // short card heading (≤4 words noun phrase)
  metric_value: number                                    // headline number (typically a percent in 0-100)
  metric_label: string                                    // axis/unit label e.g. 'CS rate at home'
  takeaway: string                                        // plain-English meaning of the metric
  action_hint: string                                     // verb-led ≤7-word recommendation
  benchmark_value: number                                 // reference line for the progress-bar (0-100)
  gw_coverage: string                                     // e.g. 'GW1–34', 'this GW', 'season-to-date'
  player_ids: number[]                                    // FPL player IDs relevant to this insight (empty if not player-specific)
  team_ids: number[]                                      // FPL team IDs relevant to this insight (empty if not team-specific)
  player_names: string[]                                  // web_name display strings, parallel to player_ids
  team_names: string[]                                    // team short_name display strings, parallel to team_ids
  // Phase 79 D-04/D-05 — signal label computed by pipeline (NOT derived client-side)
  signal_label: SignalLabel
}
