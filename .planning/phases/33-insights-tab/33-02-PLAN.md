**File 1 — Create `src/app/api/insights/route.ts` with this EXACT content:**

```typescript
import { list } from '@vercel/blob'
import { readFile } from 'fs/promises'
import { join } from 'path'

const USE_BLOB = process.env.USE_BLOB?.toLowerCase() === 'true'

export async function GET() {
  try {
    let data: string

    if (USE_BLOB) {
      const { blobs } = await list({ prefix: 'insights.json', limit: 1 })
      if (!blobs.length) {
        return Response.json({ error: 'Insights not available' }, { status: 404 })
      }
      const res = await fetch(blobs[0].url)
      data = await res.text()
    } else {
      const cachePath = join(process.cwd(), 'pipeline', 'cache', 'insights.json')
      data = await readFile(cachePath, 'utf-8')
    }

    return new Response(data, {
      status: 200,
      headers: {
        'Content-Type': 'application/json',
        'Cache-Control': 'public, s-maxage=3600, stale-while-revalidate=86400',
      },
    })
  } catch {
    return Response.json({ error: 'Failed to load insights' }, { status: 500 })
  }
}
```

This is the captain-picks route with three substitutions: `captain_picks.json` -> `insights.json` (twice — Blob prefix and local path), `Captain picks not available` -> `Insights not available`, `Failed to load captain picks` -> `Failed to load insights`. Do NOT add new imports, do NOT change cache headers, do NOT add a POST handler, do NOT add a `default` export.

```typescript
import { useQuery } from '@tanstack/react-query'
import type { Insight } from '../types'

export function useInsights() {
```

```typescript
'use client'

import { useInsights } from '@/lib/hooks/useInsights'
import type { Insight } from '@/lib/types'

// D-04 tier thresholds + D-05 badge colours (LOCKED by 33-UI-SPEC.md)
const TIER_CLASSES = {
  HIGH:   'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200',
  MEDIUM: 'bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200',
  LOW:    'bg-zinc-100 text-zinc-600 dark:bg-zinc-800 dark:text-zinc-400',
} as const

type Tier = keyof typeof TIER_CLASSES

function getTier(pct: number): Tier {
  if (pct >= 70) return 'HIGH'
  if (pct >= 50) return 'MEDIUM'
  return 'LOW'
}

// D-06 four categories — render in this fixed order.
const CATEGORY_ORDER = ['defensive', 'attacking', 'player', 'captaincy'] as const
const CATEGORY_LABELS: Record<typeof CATEGORY_ORDER[number], string> = {
  defensive: 'Defensive Patterns',
  attacking: 'Attacking Patterns',
  player:    'Player-Specific Patterns',
  captaincy: 'Captaincy Patterns',
}

function InsightCard({ insight }: { insight: Insight }) {
  const tier = getTier(insight.confidence_pct)
  const tooltip = `True in ${insight.confidence_pct}% of fixtures — ${insight.sample_n}/${insight.sample_total} matches`
  return (
    <div className="rounded border border-zinc-200 dark:border-zinc-700 bg-white dark:bg-zinc-900 p-4 space-y-2">
      <p className="text-sm">{insight.statement}</p>
      <span
        className={`inline-block text-xs font-normal rounded px-2 py-1 cursor-help ${TIER_CLASSES[tier]}`}
        title={tooltip}
      >
        {tier}
      </span>
    </div>
  )
}

export function InsightsTab() {
  const { data, isLoading, error } = useInsights()

  if (isLoading) {
    return (
      <p className="text-sm text-zinc-500 dark:text-zinc-400 text-center py-8">
        Loading insights…
      </p>
    )
  }

  if (error) {
    return (
      <p className="text-sm text-red-600 dark:text-red-400 py-4">
        Failed to load insights. Check the pipeline output and refresh.
      </p>
    )
  }

  if (!data || data.length === 0) {
    return (
      <section className="mt-6 space-y-2">
        <h2 className="text-lg font-semibold">No insights available yet</h2>
        <p className="text-sm text-zinc-500 dark:text-zinc-400">
          Run the pipeline to generate pattern data for this season.
        </p>
      </section>
    )
  }

  // Group insights by category (already sorted by pipeline as (category asc, confidence_pct desc))
  const byCategory: Record<string, Insight[]> = {
    defensive: [],
    attacking: [],
    player:    [],
    captaincy: [],
  }
  for (const insight of data) {
    if (insight.category in byCategory) {
      byCategory[insight.category].push(insight)
    }
  }

  return (
    <section className="mt-6 space-y-6">
      {CATEGORY_ORDER.map((cat) => {
        const items = byCategory[cat]
        if (items.length === 0) return null
        return (
          <div key={cat}>
            <h2 className="text-lg font-semibold mb-2">{CATEGORY_LABELS[cat]}</h2>
            <div className="space-y-3">
              {items.map((insight) => (
                <InsightCard key={insight.id} insight={insight} />
              ))}
            </div>
          </div>
        )
      })}
      <p className="text-xs text-zinc-500 dark:text-zinc-400 mt-4">
        Patterns shown only when seen in 10 or more fixtures.
      </p>
    </section>
  )
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| pipeline/cache/insights.json -> /api/insights | File reads via `readFile` with a fixed (non-user-controlled) path |
| /api/insights -> useInsights -> InsightsTab | JSON over HTTP within the same origin; React Query parses response |
| User -> InsightsTab | Read-only display; no inputs, no mutations, no destructive actions; HTML `title` attribute tooltip is browser-native |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-33-07 | Tampering | Route handler returns malformed JSON | mitigate | Route uses `new Response(data, { headers: { 'Content-Type': 'application/json', ... } })` after the cache file is read. Same primitive as `/api/captain-picks` and `/api/set-pieces` — already audited. The hook's `res.json()` will throw if the body is invalid JSON; React Query surfaces the error to the component which shows the locked error copy. |
| T-33-08 | Information Disclosure | Path traversal via filename | accept | Path is hard-coded `join(process.cwd(), 'pipeline', 'cache', 'insights.json')` — no user input, no traversal. Same as `/api/captain-picks`. |
| T-33-09 | Denial of Service | Repeated fetches thrashing the cache | mitigate | React Query `staleTime: 6 * 60 * 60 * 1000` (6 hours) prevents re-fetch unless the cache is invalidated or the page is hard-reloaded. Matches usePlayers, useSetPieces, useCaptainPicks pattern. |
| T-33-10 | Repudiation | Cache file replaced between reads | accept | `Cache-Control: public, s-maxage=3600, stale-while-revalidate=86400` on the response — same headers as other internal API routes. Stale-while-revalidate is the standard caching trade-off for daily-refresh data. |
| T-33-11 | Tampering | XSS via insight statement string | mitigate | React renders strings through JSX which escapes HTML by default. Statements are constructed in pipeline/insights.py via Python f-strings using FPL data (already curated by upstream Zod parsers in fpl_client.py) — no `dangerouslySetInnerHTML` anywhere. The badge `title` attribute is also rendered by React and is HTML-escaped. |
| T-33-12 | Tampering | XSS via tier badge tooltip | mitigate | `title={tooltip}` uses React's safe attribute interpolation — no innerHTML. Browser-native rendering of the title attribute escapes any HTML inside it. |
| T-33-13 | Spoofing | Mocked hook in tests bypasses real fetch | accept | `vi.mock()` is scoped to test files only. Production bundle uses the real hook. No risk to deployed app. |
| T-33-14 | Information Disclosure | API route serves the entire insights.json publicly | accept | All data in insights.json is derived from FPL public bootstrap-static + fixtures + element-summary. There is no PII, no auth tokens, no secrets. Same risk profile as merged_players.json, captain_picks.json, defcon_stats.json. |
