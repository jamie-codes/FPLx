```typescript
// src/lib/opportunity-cost.ts

export type OCSRowKind = 'roll' | 'single-free' | 'single-hit' | 'combo-free' | 'combo-hit'

export const MARGINAL_THRESHOLD = 1.0

export interface OCSRow {
  kind: OCSRowKind
  label: string                     // "Roll" | "1 FT" | "1 FT (Hit)" | "2 FT"
  xPtsGain: number                  // raw gain (0 for Roll)
  xPtsGainNet: number               // xPtsGain - cost  (so Hit rows are net of -4)
  xPtsGainPerGw: number             // 0 for Roll; suggestion.xPtsGainPerGw otherwise
  breakEvenGws: number | null       // null for Roll/Free; positive int for Hit
  cost: 0 | 4
  transfers?: Array<{ sell: MergedPlayer; buy: MergedPlayer }>  // undefined for Roll
  isMarginal?: boolean              // only set on combo-free when xPtsGain < MARGINAL_THRESHOLD
}

export function computeOpportunityCostRows(
  suggestions: TransferSuggestion[],
  ftCount: 1 | 2,
): OCSRow[]
```

Row ordering invariant (REQUIRED — tested):
- Index 0: always `kind: 'roll'`.
- Index 1: `single-free` when present.
- Index 2 (ftCount=1): `single-hit` when present. Index 2 (ftCount=2): `combo-free` when present.
- Rows for non-existent suggestions are simply omitted (no placeholder rows).
