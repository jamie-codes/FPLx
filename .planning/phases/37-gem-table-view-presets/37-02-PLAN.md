    ```tsx
    import type { ViewPreset } from './GwToggle'

    const PRESET_LABELS: Record<ViewPreset, string> = {
      default: 'Default',
      compact: 'Compact',
      analysis: 'Analysis',
    }

      preset: ViewPreset
      onPresetChange: (p: ViewPreset) => void
    }

    export function PresetToggle({ preset, onPresetChange }: PresetToggleProps) {
      return (
        <div
          role="group"
          aria-label="Table view preset"
          className="hidden sm:flex rounded overflow-hidden border border-zinc-300 dark:border-zinc-600"
        >
          {(['default', 'compact', 'analysis'] as const).map((p) => (
            <button
              key={p}
              onClick={() => onPresetChange(p)}
              aria-pressed={preset === p}
              className={`px-3 py-1 text-sm font-medium transition-all cursor-pointer active:scale-95 ${
                preset === p
                  ? 'bg-zinc-900 dark:bg-white text-white dark:text-zinc-900'
                  : 'bg-white dark:bg-zinc-800 text-zinc-700 dark:text-zinc-300 hover:bg-zinc-50 dark:hover:bg-zinc-700'
              }`}
            >
              {PRESET_LABELS[p]}
            </button>
          ))}
        </div>
      )
    }
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-37-04 | Tampering | PresetToggle onClick | accept | onClick only calls `onPresetChange(p)` where `p` is a compile-time const from `(['default', 'compact', 'analysis'] as const)`. No user-controlled string enters the state; TypeScript enforces `ViewPreset` type. |
| T-37-05 | Information Disclosure | Column visibility in DOM | accept | Hiding columns with TanStack `columnVisibility` removes them from the rendered DOM but the underlying data is already client-side in the `scoredPlayers` array. This is a display-only tool with no authentication; no sensitive PII is at risk. All data is already public (FPL API). |
| T-37-06 | Elevation of Privilege | gemPreset state in page.tsx | accept | State is in-memory React useState. No persistence, no server-side effects. Cannot be used to access resources or escalate permissions. |
| T-37-07 | Denial of Service | Rapid toggling causing re-renders | accept | Preset change triggers a single setState call; TanStack Table re-computes columnVisibility synchronously. No debouncing needed — the operation is O(24 keys), sub-millisecond. |
</threat_model>
