<objective>
Ship a pure-TypeScript candidate engine that takes the full `MergedPlayer[]` pool and returns the top-5 captain candidates re-ranked per the four locked sort modes from CONTEXT.md D-03. Engine is consumed in Plan 02 by the rewritten `CaptainPicksPanel`.

Purpose: Isolate the EO ranking logic in a unit-testable pure function so the four sort/filter rules (Max xPts, Protect Rank, Chase Rank, Differential Aggressive) are locked behind regression tests before the UI wiring lands. Solves Pitfall 1 (wrong candidate pool size — `useCaptainPicks` only returns 2 picks) and Pitfall 2 (median computed on already-filtered subset) by enforcing them at the lib boundary.

- `src/lib/eo-candidates.ts` exporting `computeEOCandidates(players, mode, topN=5): MergedPlayer[]` and `EOMode` type
- `src/lib/eo-candidates.test.ts` with full mode coverage in jsdom-safe vitest format

    ```typescript
    // Phase 57 (EO-01..EO-03): computeEOCandidates — pure ranker for the captain panel.
    // Sources of truth:
    //   - .planning/phases/057-effective-ownership-mode/057-CONTEXT.md §decisions D-02, D-03, D-07
    //   - .planning/phases/057-effective-ownership-mode/057-RESEARCH.md §Code Examples
    // Eligibility rule mirrors src/lib/captaincy-engine.ts lines 70–78 plus an extra status === 'a' gate.
    import type { MergedPlayer } from '@/lib/types'

    export type EOMode = 'max_xpts' | 'protect_rank' | 'chase_rank' | 'differential_aggressive'

    export function computeEOCandidates(
      players: MergedPlayer[],
      mode: EOMode,
      topN = 5,
    ): MergedPlayer[] {
      const eligible = players.filter(
        p =>
          p.status === 'a' &&
          p.element_type !== 1 &&
          p.xPts_1gw != null &&
          p.xPts_1gw > 0,
      )

      if (mode === 'max_xpts') {
        return eligible
          .slice()
          .sort((a, b) => (b.xPts_1gw ?? 0) - (a.xPts_1gw ?? 0))
          .slice(0, topN)
      }

      if (mode === 'protect_rank') {
        return eligible
          .slice()
          .sort(
            (a, b) =>
              parseFloat(b.selected_by_percent) -
              parseFloat(a.selected_by_percent),
          )
          .slice(0, topN)
      }

      if (mode === 'chase_rank') {
        return eligible
          .slice()
          .sort((a, b) => (b.xPts_90th_1gw ?? 0) - (a.xPts_90th_1gw ?? 0))
          .slice(0, topN)
      }

      // differential_aggressive — Pitfall 2: median over the FULL eligible pool.
      const xptsValues = eligible
        .map(p => p.xPts_1gw ?? 0)
        .slice()
        .sort((a, b) => a - b)
      const mid = Math.floor(xptsValues.length / 2)
      const median =
        xptsValues.length === 0
          ? 0
          : xptsValues.length % 2 !== 0
            ? xptsValues[mid]
            : (xptsValues[mid - 1] + xptsValues[mid]) / 2
      return eligible
        .filter(p => (p.xPts_1gw ?? 0) >= median)
        .slice()
        .sort(
          (a, b) =>
            parseFloat(a.selected_by_percent) -
            parseFloat(b.selected_by_percent),
        )
        .slice(0, topN)
    }
    ```
