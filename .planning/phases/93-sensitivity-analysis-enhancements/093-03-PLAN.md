    ```typescript
    // Phase 93 (SENS-01): FragilityBadge — tristate inline indicator.
    // Sources of truth:
    //   - .planning/phases/93-sensitivity-analysis-enhancements/93-UI-SPEC.md §Component Specification: FragilityBadge
    //   - .planning/phases/93-sensitivity-analysis-enhancements/93-CONTEXT.md §decisions D-07, D-08, D-09
    // Tier mapping:
    //   'robust'     → renders nothing (null)
    //   'fragile'    → amber tone:  text-amber-600 dark:text-amber-400
    //   'knife_edge' → orange tone: text-orange-600 dark:text-orange-400
    // Pitfall 4 (from Phase 64): NO filled-pill classes (bg-*, inline-block, rounded) — preserves
    // visual distinction from DangerousToFadeBadge / McLabel / SeverityBadge MEDIUM / RotationRiskBadge.

    import type { FragilityTier } from '@/lib/sensitivity'

    interface FragilityBadgeProps {
      tier: FragilityTier
      reasons: string[]
    }

    const TIER_CLASSES: Record<Exclude<FragilityTier, 'robust'>, string> = {
      fragile:     'text-xs text-amber-600 dark:text-amber-400',
      knife_edge:  'text-xs text-orange-600 dark:text-orange-400',
    }

    export function FragilityBadge({ tier, reasons }: FragilityBadgeProps) {
      if (tier === 'robust') return null
      return (
        <div className={TIER_CLASSES[tier]} data-testid="fragility-badge">
          <span aria-hidden="true">⚠ </span>
          {`no longer recommended if: ${reasons.join(', ')}`}
        </div>
      )
    }
    ```

    Notes on the implementation:
    - `TIER_CLASSES` is a `Record<Exclude<FragilityTier, 'robust'>, string>` — the `Exclude<..., 'robust'>` makes the lookup type-safe AT compile time (the type system knows `tier` is not `'robust'` after the early-return guard).
    - The early `return null` for `'robust'` happens BEFORE any reason rendering; the reasons array is irrelevant in this branch.
    - The `<span aria-hidden="true">` is identical to `FragilityNote` — screen readers skip the ⚠ glyph and read the prose.
    - The copy template `'no longer recommended if: ...'` is identical to `FragilityNote` — same prefix, same comma-join. Reasons.length === 0 with `tier !== 'robust'` is a degenerate state that should not occur (the engine in 093-02 never returns `tier: 'fragile'` with empty reasons), but if it did, the badge would render `'no longer recommended if: '` with a trailing space — acceptable since callsites use `tier !== 'robust'` as the render gate, not `reasons.length`.

    Do NOT add:
    - No `'use client'` directive — `FragilityNote.tsx` does not have one and Next.js will treat this as a server component fragment (or hoist it as needed). If the build fails because a parent is a server component, the parent already declares `'use client'` (CaptainPicksPanel, GemTable, OpportunityCostTable all have `'use client'` at top). Component itself is presentational — no hooks, no event handlers — so directive is unnecessary.
    - No additional props beyond `tier` and `reasons`.
    - No tooltip, no `title` attribute, no `role` attribute.
    - No CSS modules, no Tailwind variant classes beyond the two declared in `TIER_CLASSES`.

    ```typescript
    // Phase 93 (SENS-01): FragilityBadge — RTL component tests.
    // @vitest-environment jsdom
    import { describe, it, expect } from 'vitest'
    import { render } from '@testing-library/react'
    import { FragilityBadge } from './FragilityBadge'

    describe('FragilityBadge — Phase 93 SENS-01', () => {
      it('renders nothing when tier is robust (empty reasons)', () => {
        const { container } = render(<FragilityBadge tier="robust" reasons={[]} />)
        expect(container.firstChild).toBeNull()
        expect(container.querySelector('[data-testid="fragility-badge"]')).toBeNull()
      })

      it('renders nothing when tier is robust even with non-empty reasons (D-07)', () => {
        const { container } = render(
          <FragilityBadge tier="robust" reasons={['start_prob < 70%']} />
        )
        expect(container.firstChild).toBeNull()
      })

      it('renders amber badge for fragile tier with single reason', () => {
        const { container } = render(
          <FragilityBadge tier="fragile" reasons={['start_prob < 70%']} />
        )
        const badge = container.querySelector('[data-testid="fragility-badge"]')
        expect(badge).not.toBeNull()
        expect(badge?.textContent ?? '').toContain('no longer recommended if: start_prob < 70%')
        const cls = badge?.className ?? ''
        expect(cls).toContain('text-xs')
        expect(cls).toContain('text-amber-600')
        expect(cls).toContain('dark:text-amber-400')
      })

      it('fragile tier MUST NOT include filled-pill classes (Pitfall 4)', () => {
        const { container } = render(
          <FragilityBadge tier="fragile" reasons={['harder fixture']} />
        )
        const cls = container.querySelector('[data-testid="fragility-badge"]')?.className ?? ''
        expect(cls).not.toContain('bg-amber-100')
        expect(cls).not.toContain('bg-amber-900')
        expect(cls).not.toContain('inline-block')
        expect(cls).not.toContain('rounded')
      })

      it('renders orange badge for knife_edge tier with multiple reasons', () => {
        const { container } = render(
          <FragilityBadge
            tier="knife_edge"

          />

        expect(cls).toContain('text-orange-600')
        expect(cls).toContain('dark:text-orange-400')
        expect(cls).not.toContain('text-amber-600')
        expect(cls).not.toContain('text-amber-400')
      })

      it('knife_edge tier MUST NOT include filled-pill classes (Pitfall 4)', () => {

        expect(cls).not.toContain('bg-orange-100')
        expect(cls).not.toContain('bg-orange-900')
        expect(cls).not.toContain('inline-block')
        expect(cls).not.toContain('rounded')
      })

        for (const tier of ['fragile', 'knife_edge'] as const) {

          )

          expect(ariaHiddenSpan).not.toBeNull()

          const matches = text.match(/no longer recommended if:/g) ?? []
          expect(matches.length).toBe(1)
        }
      })
    })
    ```
