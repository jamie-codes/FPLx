This file defines the TanStack Table column configuration. It does NOT need `'use client'` — it is imported by GemTable.tsx which has the directive.

```typescript
import { createColumnHelper } from '@tanstack/react-table'
import type { ScoredPlayer } from '@/lib/types'

const col = createColumnHelper<ScoredPlayer>()

// Position code to label map
const POS_LABEL: Record<number, string> = { 1: 'GK', 2: 'DEF', 3: 'MID', 4: 'FWD' }

// Format score as 0-100 integer for display
const fmtScore = (v: number) => (v * 100).toFixed(0)
const fmtScoreNull = (v: number | null) => (v === null ? '\u2014' : (v * 100).toFixed(0))
const fmtDec2 = (v: number | null) => (v === null ? '\u2014' : v.toFixed(2))

export const columns = [
  col.accessor('web_name', { header: 'Player', enableSorting: true }),
  col.accessor('team_short_name', { header: 'Team', enableSorting: false }),
  col.accessor('element_type', {
    header: 'Pos',
    filterFn: 'equals',
    enableSorting: false,
    cell: (info) => POS_LABEL[info.getValue()] ?? '?',
  }),
  col.accessor('now_cost', {
    header: 'Price',
    cell: (info) => `${(info.getValue() / 10).toFixed(1)}`,
  }),
  col.accessor('gem_score', {
    header: 'Gem',
    cell: (info) => fmtScore(info.getValue()),
  }),
  col.accessor('fdr_score', {
    header: 'FDR',
    cell: (info) => fmtScore(info.getValue()),
  }),
  col.accessor('form_score', {
    header: 'Form',
    cell: (info) => fmtScore(info.getValue()),
  }),
  col.accessor('xg_per90', {
    header: 'xG/90',
    cell: (info) => fmtDec2(info.getValue()),
  }),
  col.accessor('xa_per90', {
    header: 'xA/90',
    cell: (info) => fmtDec2(info.getValue()),
  }),
  col.accessor('xg_score', {
    header: 'xG Sc',
    cell: (info) => fmtScoreNull(info.getValue()),
  }),
  col.accessor('xa_score', {
    header: 'xA Sc',
    cell: (info) => fmtScoreNull(info.getValue()),
  }),
  col.accessor('ownership_score', {
    header: 'Own',
    cell: (info) => fmtScore(info.getValue()),
  }),
  col.accessor('minutes_score', {
    header: 'Min',
    cell: (info) => fmtScore(info.getValue()),
  }),
  col.accessor('set_piece_score', {
    header: 'SP',
    cell: (info) => fmtScore(info.getValue()),
  }),
  col.accessor('selected_by_percent', { header: 'Own%' }),
  col.accessor('status', {
    header: 'Status',
    enableSorting: false,
    cell: (info) => info.getValue() === 'a' ? '' : info.getValue().toUpperCase(),
  }),
]
```

**2. Create `src/components/gem-table/PositionFilter.tsx`:**

```typescript
'use client'

import type { PositionCode } from '@/lib/types'

const POSITIONS: Array<{ label: string; code: PositionCode | null }> = [
  { label: 'All', code: null },
  { label: 'GK', code: 1 },
  { label: 'DEF', code: 2 },
  { label: 'MID', code: 3 },
  { label: 'FWD', code: 4 },
]

interface Props {
  active: PositionCode | null
  onChange: (code: PositionCode | null) => void
}

export function PositionFilter({ active, onChange }: Props) {
  return (
    <div className="flex gap-2 mb-4">
      {POSITIONS.map(({ label, code }) => (
        <button
          key={label}
          onClick={() => onChange(code)}
          className={`px-3 py-1 rounded text-sm font-medium transition-colors ${
            active === code
              ? 'bg-blue-600 text-white'
              : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
          }`}
        >
          {label}
        </button>
      ))}
    </div>
  )
}
```

- gem_score displayed as 0-100 integer (not 0.0-1.0 raw float) for readability

**1. Create `src/components/gem-table/GemTable.tsx`:**

This is the main 'use client' component. It:
- Calls `usePlayers()` to get MergedPlayer[] data
- Calls `computeAllGemScores(data)` to produce ScoredPlayer[] (memo'd with useMemo)
- Manages `SortingState` with default `[{ id: 'gem_score', desc: true }]`
- Manages `ColumnFiltersState` (driven by PositionFilter)
- Calls `useReactTable` with `getCoreRowModel`, `getSortedRowModel`, `getFilteredRowModel`
- Renders loading/error states
- Renders PositionFilter, row count, and the table

Structure:
```typescript
'use client'

import { useState, useMemo } from 'react'
import {
  flexRender,
  getCoreRowModel,
  getFilteredRowModel,
  getSortedRowModel,
  useReactTable,
  type ColumnFiltersState,
  type SortingState,
} from '@tanstack/react-table'
import { usePlayers } from '@/lib/hooks/usePlayers'
import { computeAllGemScores } from '@/lib/gem-score'
import type { PositionCode } from '@/lib/types'
import { columns } from './columns'
import { PositionFilter } from './PositionFilter'
```

Key implementation details:
- `useMemo(() => computeAllGemScores(data ?? []), [data])` to avoid recomputing on every render
- Track active position as `PositionCode | null` state, pass to PositionFilter
- When position changes, set `columnFilters` to `[{ id: 'element_type', value: code }]` or `[]` for All
- Show `table.getRowModel().rows.length` as row count below filter
- Loading state: `"Loading players..."` text
- Error state: `"Failed to load players"` with error message
- Table styling with Tailwind:
  - `<table className="w-full text-sm text-left">`
  - `<th>` with `cursor-pointer` when sortable, sort direction indicator (triangle characters: `\u25B2` for asc, `\u25BC` for desc)
  - Alternating row colours: `even:bg-gray-50`
  - Sticky header: `sticky top-0 bg-white`
  - Compact padding: `px-2 py-1`
- Page title: `<h1 className="text-2xl font-bold mb-4">Gem Ratings</h1>`
